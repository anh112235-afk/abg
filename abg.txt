#include<stdio.h>
#include<stdlib.h>

typedef struct {
	char HoTen[30];
	int masv;
	float d1 , d2 ,d3;
}Data;

typedef struct tagNode {
	Data infor;
	struct tagNode *link;
}Node;

typedef struct {
	Node *pHead;
	Node *pTail;
	int spt;
}LList;
void taoList(LList *L){
	L->pHead = NULL;
	L->pTail = NULL;
	L->spt = 0;
}

int emptyList(LList L){
	return (L.spt == 0);
}
//1
void themDau (LList *L , Data x){
	Node *pp;
	pp = (Node*)malloc(sizeof(Node));
	pp->infor = x;
	pp->link = NULL;
	
	if  ( emptyList(*L)){
		L->pHead = pp;
		L->pTail == pp;
	}
	else {
		pp->link = L->pHead;
		L->pHead = pp;
	}
	L->spt++;
}
//2
void themCuoi (LList *L , Data x){
	Node *pp;
	pp = (Node*)malloc(sizeof(Node));
	pp->infor = x;
	pp->link =NULL;
	if(emptyList(*L)){
		L->pHead= pp;
		L->pTail == pp;
	}
	else{
		L->pTail->link = pp;
		L->pTail = pp;
	}
	L->spt++;
}
//3
void themSau(LList *L ,Node *q, Data x){
	Node *pp;
	if (q == NULL) return;
	else {
		pp = (Node*)malloc(sizeof(Node));
		pp->infor = x;
		pp->link = q->link;
		q-link = pp;
		if(q == L->pTail) L->pTail == pp;
		L->spt++;
	}
}

void chen_tang(LList *L , Data x){
	Node *pp = L->pHead, q*;
	while (pp != NULL && pp->infor.masv < x.masv){
		q =pp;
		pp = pp->link;
	}
	if(pp == L->pHead) themDau(L,x);
	else themSau(L,q,x);
}

void tach(LList L , LList *L1 , LList *L2){
	taoList(L1);
	taoList(L2);
	Node *pp = L.pHead;
	while (pp != NULL){
		if(pp->infor.d1 < 7.0 || pp->infor.d2 < 7.0|| pp->infor.d3 < 7.0){
			themDau(L1, pp->infor);
		}else themDau(L2,pp->infor);
		pp == pp->link;
	}
}

void ghep(LList L1 , LList L2 , LList *L){
	taoList(L);
	Node *p1 = L1.pHead;
	Node *p2 = L2.pHead;
	
	if(L1.spt < L2.spt){
		while(p1 != NULL){
			themDau(L , p1->infor);
			themDau(L,p2->infor);
			p1 = p1->link;
			p2 = p2->link;
		}
		while(p2 != NULL){
			themDau(L, p2->infor);
			p2 = p2->link;
		}
	}else {
		while ( p2 != NULL){
			themDau(L , p2->infor);
			themDau(L, p1->infor);
			p2 = p2->link;
			p1 = p1->link;
		}
		while ( p1 != NULL){
			themDau(L , p1->infor);
			p1 = p1->link;
		}
	}
}
//7
void xoaDau(LList *L){
	Node *pp;
	if(emptyList(*L)) return;
	else {
		pp = L->pHead;
		L->pHead = pp->link;
		free(pp);
		L->spt--;
		if(L->pHead == NULL) L->pTail = NULL;
	}
}
//8
void xoaCuoi (LList *L , Node *q){
	Node *pp;
	if(q == NULL) return;
	else {
		pp = q->link;
		if(q->link == L->pTail)
			L->pTail = q;
		q->link = pp->link;
		free(pp);
		L->spt--;
	}
}

void xulyList(LList L){
	Node *pp = L.pHead;

		in(pp->infor);
		pp = pp->link;
